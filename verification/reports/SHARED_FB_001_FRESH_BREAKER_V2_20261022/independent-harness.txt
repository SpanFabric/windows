def digest(repo):
 h=hashlib.sha256()
 raw=subprocess.check_output(['git','ls-files','-s','-z'],cwd=repo)
 for rec in sorted(raw.split(b'\0'),key=lambda x:x.split(b'\t')[-1]):
  if not rec:continue
  meta,path=rec.split(b'\t');mode,oid,stage=meta.split()
  if path==b'verification/state.yaml' or path.startswith(b'verification/reports/'):continue
  data=subprocess.check_output(['git','cat-file','blob',oid.decode()],cwd=repo)
  for label,val in [(b'path',path),(b'mode',mode),(b'index_oid',oid),(b'type',b'regular'),(b'length',str(len(data)).encode()),(b'content',data)]:h.update(label+b'\0'+val+b'\0')
 return h.hexdigest()
